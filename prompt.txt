
cd ~/AquaCulture

bash <<'AQUACULTURE_SETUP'
set -Eeuo pipefail

PROJECT_DIR="$HOME/AquaCulture"
APP_USER="$(id -un)"
VENV_DIR="$PROJECT_DIR/.venv"
PYTHON="$VENV_DIR/bin/python"

WEB_SERVICE="aquaculture-web.service"
BLE_SERVICE="aquaculture-ble.service"

echo "=========================================="
echo " AQUACULTURE AUTOMATIC STARTUP SETUP"
echo "=========================================="
echo "Project: $PROJECT_DIR"
echo "User: $APP_USER"
echo

# ------------------------------------------
# 1. Validate project files
# ------------------------------------------

cd "$PROJECT_DIR"

for file in \
    "server/app.py" \
    "server/requirements.txt" \
    "network/rpi/main.py" \
    "network/rpi/requirements.txt" \
    "startup/install.sh"; do

    if [ ! -f "$file" ]; then
        echo "ERROR: Missing $file"
        exit 1
    fi
done

echo "[OK] Project structure verified."

# ------------------------------------------
# 2. Stop existing services
# ------------------------------------------

echo "[INFO] Stopping existing services..."

sudo systemctl stop \
    "$WEB_SERVICE" \
    "$BLE_SERVICE" 2>/dev/null || true

# ------------------------------------------
# 3. Install operating system packages
# ------------------------------------------

echo "[INFO] Installing system dependencies..."

sudo apt-get update

sudo apt-get install -y \
    python3 \
    python3-pip \
    python3-venv \
    python3-dev \
    python3-dbus \
    python3-gi \
    python3-gi-cairo \
    gir1.2-glib-2.0 \
    build-essential \
    pkg-config \
    libglib2.0-dev \
    libdbus-1-dev \
    libbluetooth-dev \
    bluetooth \
    bluez \
    network-manager

echo "[OK] System dependencies installed."

# ------------------------------------------
# 4. Create Python virtual environment
# ------------------------------------------

echo "[INFO] Configuring Python environment..."

if [ ! -f "$PYTHON" ]; then
    python3 -m venv \
        --system-site-packages \
        "$VENV_DIR"
fi

# Allow access to OS-provided D-Bus/PyGObject
# without attempting to reinstall them via pip.
if [ -f "$VENV_DIR/pyvenv.cfg" ]; then
    sed -i \
        's/^include-system-site-packages = false/include-system-site-packages = true/' \
        "$VENV_DIR/pyvenv.cfg"
fi

"$PYTHON" -m pip --version

echo "[OK] Python environment ready."

# ------------------------------------------
# 5. Install Python dependencies
# ------------------------------------------

echo "[INFO] Installing Flask backend packages..."

"$PYTHON" -m pip install \
    -r "$PROJECT_DIR/server/requirements.txt"

echo "[INFO] Installing Bluetooth packages..."

"$PYTHON" -m pip install \
    -r "$PROJECT_DIR/network/rpi/requirements.txt"

# ------------------------------------------
# 6. Verify essential Python modules
# ------------------------------------------

echo "[INFO] Checking Python modules..."

"$PYTHON" - <<'PYTHON_CHECK'
import importlib

modules = [
    "flask",
    "dbus",
    "gi",
]

for name in modules:
    importlib.import_module(name)
    print(f"[OK] {name}")

print("[OK] Essential Python imports succeeded.")
PYTHON_CHECK

# ------------------------------------------
# 7. Enable operating system services
# ------------------------------------------

echo "[INFO] Enabling Bluetooth service..."

sudo systemctl enable bluetooth.service
sudo systemctl start bluetooth.service

# NetworkManager is required by AquaCulture
# Bluetooth Wi-Fi provisioning.
# Do not automatically replace an existing
# network configuration manager.
if systemctl is-active --quiet NetworkManager.service; then
    echo "[OK] NetworkManager is running."
else
    echo "[WARNING] NetworkManager is not running."
    echo "[WARNING] Bluetooth Wi-Fi provisioning may need"
    echo "          separate network configuration."
fi

# ------------------------------------------
# 8. Install AquaCulture startup services
# ------------------------------------------

echo "[INFO] Installing AquaCulture services..."

sudo bash "$PROJECT_DIR/startup/install.sh"

# ------------------------------------------
# 9. Configure Python executable via
#    persistent systemd overrides
# ------------------------------------------

echo "[INFO] Configuring Python service paths..."

sudo mkdir -p \
    "/etc/systemd/system/$WEB_SERVICE.d" \
    "/etc/systemd/system/$BLE_SERVICE.d"

printf '[Service]\nExecStart=\nExecStart=%s -u %s/server/app.py\n' \
    "$PYTHON" "$PROJECT_DIR" |
    sudo tee \
        "/etc/systemd/system/$WEB_SERVICE.d/override.conf" \
        >/dev/null

printf '[Service]\nExecStart=\nExecStart=%s -u %s/network/rpi/main.py\n' \
    "$PYTHON" "$PROJECT_DIR" |
    sudo tee \
        "/etc/systemd/system/$BLE_SERVICE.d/override.conf" \
        >/dev/null

sudo systemctl daemon-reload

# ------------------------------------------
# 10. Enable automatic startup
# ------------------------------------------

echo "[INFO] Enabling AquaCulture on boot..."

sudo systemctl enable \
    "$WEB_SERVICE" \
    "$BLE_SERVICE"

# ------------------------------------------
# 11. Restart AquaCulture services
# ------------------------------------------

echo "[INFO] Starting AquaCulture..."

sudo systemctl restart "$WEB_SERVICE"
sudo systemctl restart "$BLE_SERVICE"

echo "[INFO] Waiting for startup..."
sleep 15

# ------------------------------------------
# 12. Verify service status
# ------------------------------------------

echo
echo "=========================================="
echo " SERVICE STATUS"
echo "=========================================="

systemctl --no-pager --full status \
    "$WEB_SERVICE" || true

systemctl --no-pager --full status \
    "$BLE_SERVICE" || true

echo
echo "=========================================="
echo " AUTOMATIC STARTUP"
echo "=========================================="

systemctl is-enabled \
    "$WEB_SERVICE" \
    "$BLE_SERVICE" || true

echo
echo "=========================================="
echo " SERVICE LOGS"
echo "=========================================="

sudo journalctl \
    -u "$WEB_SERVICE" \
    -n 25 \
    --no-pager

sudo journalctl \
    -u "$BLE_SERVICE" \
    -n 25 \
    --no-pager

echo
echo "=========================================="
echo " FINAL VERIFICATION"
echo "=========================================="

FAILED=0

for SERVICE in "$WEB_SERVICE" "$BLE_SERVICE"; do

    if systemctl is-active --quiet "$SERVICE"; then
        echo "[OK] $SERVICE is running."
    else
        echo "[FAILED] $SERVICE is not running."
        FAILED=1
    fi

    if systemctl is-enabled --quiet "$SERVICE"; then
        echo "[OK] $SERVICE is enabled on boot."
    else
        echo "[FAILED] $SERVICE is not enabled."
        FAILED=1
    fi

done

echo
echo "Project: $PROJECT_DIR"
echo "Python: $PYTHON"
echo

if [ "$FAILED" -eq 0 ]; then
    echo "[OK] Both services are active and enabled."
    echo "Check logs and functionality before reboot testing."
else
    echo "[WARNING] One or more service checks failed."
    echo "Review the logs above before rebooting."
    exit 1
fi

AQUACULTURE_SETUP
